```go
package main

import (
	"errors"
	"fmt"
)

type Status int

const (
	Draft Status = iota
	Pending
	Approved
	Rejected
)

func (s Status) String() string {
	switch s {
	case Draft:
		return "下書き"
	case Pending:
		return "申請中"
	case Approved:
		return "承認済み"
	case Rejected:
		return "差戻し"
	default:
		return "不明"
	}
}

type Application struct {
	ID          int
	ApplicantID int
	Status      Status
}

var (
	ErrInvalidTransition  = errors.New("不正な状態遷移です")
	ErrSameApplicant      = errors.New("承認者は申請者と同一人物であってはなりません")
)

func Submit(app *Application, applicantID int) error {
	if app.Status != Draft && app.Status != Rejected {
		return fmt.Errorf("%w: %s から 申請中 への遷移はできません", ErrInvalidTransition, app.Status)
	}
	app.ApplicantID = applicantID
	app.Status = Pending
	return nil
}

func Approve(app *Application, approverID int) error {
	if app.Status != Pending {
		return fmt.Errorf("%w: %s から 承認済み への遷移はできません", ErrInvalidTransition, app.Status)
	}
	if app.ApplicantID == approverID {
		return ErrSameApplicant
	}
	app.Status = Approved
	return nil
}

func Reject(app *Application, approverID int) error {
	if app.Status != Pending {
		return fmt.Errorf("%w: %s から 差戻し への遷移はできません", ErrInvalidTransition, app.Status)
	}
	if app.ApplicantID == approverID {
		return ErrSameApplicant
	}
	app.Status = Rejected
	return nil
}

func main() {
	app := &Application{ID: 1, ApplicantID: 1, Status: Draft}
	fmt.Printf("初期状態: %s\n", app.Status)

	if err := Submit(app, 1); err != nil {
		fmt.Printf("エラー: %v\n", err)
	} else {
		fmt.Printf("提出後: %s\n", app.Status)
	}

	if err := Approve(app, 1); err != nil {
		fmt.Printf("エラー: %v\n", err)
	}

	if err := Approve(app, 2); err != nil {
		fmt.Printf("エラー: %v\n", err)
	} else {
		fmt.Printf("承認後: %s\n", app.Status)
	}

	app2 := &Application{ID: 2, ApplicantID: 1, Status: Draft}
	Submit(app2, 1)
	fmt.Printf("\n申請2 提出後: %s\n", app2.Status)

	if err := Reject(app2, 2); err != nil {
		fmt.Printf("エラー: %v\n", err)
	} else {
		fmt.Printf("差戻し後: %s\n", app2.Status)
	}

	if err := Submit(app2, 1); err != nil {
		fmt.Printf("エラー: %v\n", err)
	} else {
		fmt.Printf("再提出後: %s\n", app2.Status)
	}

	if err := Submit(app, 1); err != nil {
		fmt.Printf("\n不正遷移エラー: %v\n", err)
	}
}
```
