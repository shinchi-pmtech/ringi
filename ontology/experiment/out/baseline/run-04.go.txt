```go
package main

import (
	"errors"
	"fmt"
)

type Status int

const (
	Draft Status = iota
	Pending
	Approved
	Rejected
)

func (s Status) String() string {
	switch s {
	case Draft:
		return "下書き"
	case Pending:
		return "申請中"
	case Approved:
		return "承認済み"
	case Rejected:
		return "差戻し"
	default:
		return "不明"
	}
}

type UserID string

type Application struct {
	ID          string
	ApplicantID UserID
	Status      Status
}

var (
	ErrInvalidTransition    = errors.New("無効な状態遷移です")
	ErrSameApprover         = errors.New("承認者と申請者が同一人物であってはなりません")
	ErrNotApplicant         = errors.New("申請者のみが操作できます")
)

func Submit(app *Application, applicantID UserID) error {
	if app.Status != Draft {
		return fmt.Errorf("%w: %s -> 申請中", ErrInvalidTransition, app.Status)
	}
	if app.ApplicantID != applicantID {
		return ErrNotApplicant
	}
	app.Status = Pending
	return nil
}

func Approve(app *Application, approverID UserID) error {
	if app.Status != Pending {
		return fmt.Errorf("%w: %s -> 承認済み", ErrInvalidTransition, app.Status)
	}
	if app.ApplicantID == approverID {
		return ErrSameApprover
	}
	app.Status = Approved
	return nil
}

func Reject(app *Application, approverID UserID) error {
	if app.Status != Pending {
		return fmt.Errorf("%w: %s -> 差戻し", ErrInvalidTransition, app.Status)
	}
	if app.ApplicantID == approverID {
		return ErrSameApprover
	}
	app.Status = Rejected
	return nil
}

func main() {
	applicant := UserID("user-1")
	approver := UserID("user-2")

	app := &Application{
		ID:          "app-001",
		ApplicantID: applicant,
		Status:      Draft,
	}

	fmt.Printf("初期状態: %s\n", app.Status)

	if err := Submit(app, applicant); err != nil {
		fmt.Printf("提出エラー: %v\n", err)
	} else {
		fmt.Printf("提出後: %s\n", app.Status)
	}

	if err := Approve(app, applicant); err != nil {
		fmt.Printf("承認エラー: %v\n", err)
	}

	if err := Approve(app, approver); err != nil {
		fmt.Printf("承認エラー: %v\n", err)
	} else {
		fmt.Printf("承認後: %s\n", app.Status)
	}

	app2 := &Application{
		ID:          "app-002",
		ApplicantID: applicant,
		Status:      Draft,
	}

	_ = Submit(app2, applicant)
	fmt.Printf("app2 提出後: %s\n", app2.Status)

	if err := Reject(app2, approver); err != nil {
		fmt.Printf("差戻しエラー: %v\n", err)
	} else {
		fmt.Printf("差戻し後: %s\n", app2.Status)
	}

	if err := Approve(app2, approver); err != nil {
		fmt.Printf("無効な遷移エラー: %v\n", err)
	}
}
```
