```go
package main

import (
	"errors"
	"fmt"
)

type Status int

const (
	Draft Status = iota
	Pending
	Approved
	Rejected
)

func (s Status) String() string {
	switch s {
	case Draft:
		return "下書き"
	case Pending:
		return "申請中"
	case Approved:
		return "承認済み"
	case Rejected:
		return "差戻し"
	default:
		return "不明"
	}
}

type Application struct {
	ID          int
	ApplicantID int
	Status      Status
}

var (
	ErrInvalidTransition  = errors.New("不正な状態遷移です")
	ErrSameApplicant      = errors.New("承認者と申請者が同一人物です")
)

// Submit 下書きの申請を提出する（申請者が行う）
func Submit(app *Application) error {
	if app.Status != Draft {
		return fmt.Errorf("%ww: %s から 申請中 への遷移はできません", ErrInvalidTransition, app.Status)
	}
	app.Status = Pending
	return nil
}

// Approve 申請中の申請を承認する（承認者が行う）
func Approve(app *Application, approverID int) error {
	if app.Status != Pending {
		return fmt.Errorf("%w: %s から 承認済み への遷移はできません", ErrInvalidTransition, app.Status)
	}
	if app.ApplicantID == approverID {
		return ErrSameApplicant
	}
	app.Status = Approved
	return nil
}

// Reject 申請中の申請を差戻しする（承認者が行う）
func Reject(app *Application, approverID int) error {
	if app.Status != Pending {
		return fmt.Errorf("%w: %s から 差戻し への遷移はできません", ErrInvalidTransition, app.Status)
	}
	if app.ApplicantID == approverID {
		return ErrSameApplicant
	}
	app.Status = Rejected
	return nil
}

// Resubmit 差戻しの申請を再提出する（申請者が行う）
func Resubmit(app *Application) error {
	if app.Status != Rejected {
		return fmt.Errorf("%w: %s から 申請中 への遷移はできません", ErrInvalidTransition, app.Status)
	}
	app.Status = Pending
	return nil
}

func main() {
	app := &Application{
		ID:          1,
		ApplicantID: 1,
		Status:      Draft,
	}

	fmt.Println("=== 正常系 ===")
	fmt.Printf("初期状態: %s\n", app.Status)

	if err := Submit(app); err != nil {
		fmt.Printf("エラー: %v\n", err)
	} else {
		fmt.Printf("提出後: %s\n", app.Status)
	}

	if err := Reject(app, 2); err != nil {
		fmt.Printf("エラー: %v\n", err)
	} else {
		fmt.Printf("差戻し後: %s\n", app.Status)
	}

	if err := Resubmit(app); err != nil {
		fmt.Printf("エラー: %v\n", err)
	} else {
		fmt.Printf("再提出後: %s\n", app.Status)
	}

	if err := Approve(app, 2); err != nil {
		fmt.Printf("エラー: %v\n", err)
	} else {
		fmt.Printf("承認後: %s\n", app.Status)
	}

	fmt.Println("\n=== 異常系 ===")

	app2 := &Application{
		ID:          2,
		ApplicantID: 1,
		Status:      Draft,
	}

	// 下書きから承認しようとする
	if err := Approve(app2, 2); err != nil {
		fmt.Printf("不正な遷移エラー: %v\n", err)
	}

	// 申請者と承認者が同一
	if err := Submit(app2); err != nil {
		fmt.Printf("エラー: %v\n", err)
	}
	if err := Approve(app2, 1); err != nil {
		fmt.Printf("同一人物エラー: %v\n", err)
	}

	// 承認済みから再提出しようとする
	app3 := &Application{
		ID:          3,
		ApplicantID: 1,
		Status:      Approved,
	}
	if err := Resubmit(app3); err != nil {
		fmt.Printf("不正な遷移エラー: %v\n", err)
	}
}
```
