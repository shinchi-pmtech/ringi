```go
package main

import (
	"errors"
	"fmt"
)

type Status int

const (
	StatusDraft Status = iota
	StatusPending
	StatusApproved
	StatusRejected
)

func (s Status) String() string {
	switch s {
	case StatusDraft:
		return "下書き"
	case StatusPending:
		return "申請中"
	case StatusApproved:
		return "承認済み"
	case StatusRejected:
		return "差戻し"
	default:
		return "不明"
	}
}

type Application struct {
	ID          int
	ApplicantID int
	Status      Status
}

var (
	ErrInvalidTransition    = errors.New("invalid status transition")
	ErrSameApplicantApprover = errors.New("approver must be different from applicant")
)

func Submit(app *Application, applicantID int) error {
	if app.ApplicantID != applicantID {
		return fmt.Errorf("only the applicant can submit: %w", ErrInvalidTransition)
	}
	if app.Status != StatusDraft {
		return fmt.Errorf("cannot submit from status %s: %w", app.Status, ErrInvalidTransition)
	}
	app.Status = StatusPending
	return nil
}

func Approve(app *Application, approverID int) error {
	if app.ApplicantID == approverID {
		return fmt.Errorf("approver (id=%d) is the same as applicant: %w", approverID, ErrSameApplicantApprover)
	}
	if app.Status != StatusPending {
		return fmt.Errorf("cannot approve from status %s: %w", app.Status, ErrInvalidTransition)
	}
	app.Status = StatusApproved
	return nil
}

func Reject(app *Application, approverID int) error {
	if app.ApplicantID == approverID {
		return fmt.Errorf("approver (id=%d) is the same as applicant: %w", approverID, ErrSameApplicantApprover)
	}
	if app.Status != StatusPending {
		return fmt.Errorf("cannot reject from status %s: %w", app.Status, ErrInvalidTransition)
	}
	app.Status = StatusRejected
	return nil
}

func Resubmit(app *Application, applicantID int) error {
	if app.ApplicantID != applicantID {
		return fmt.Errorf("only the applicant can resubmit: %w", ErrInvalidTransition)
	}
	if app.Status != StatusRejected {
		return fmt.Errorf("cannot resubmit from status %s: %w", app.Status, ErrInvalidTransition)
	}
	app.Status = StatusPending
	return nil
}

func main() {
	app := &Application{
		ID:          1,
		ApplicantID: 1,
		Status:      StatusDraft,
	}

	fmt.Printf("初期状態: %s\n", app.Status)

	if err := Submit(app, 1); err != nil {
		fmt.Printf("提出エラー: %v\n", err)
	} else {
		fmt.Printf("提出後: %s\n", app.Status)
	}

	if err := Approve(app, 1); err != nil {
		fmt.Printf("承認エラー (同一人物): %v\n", err)
	}

	if err := Reject(app, 2); err != nil {
		fmt.Printf("差戻しエラー: %v\n", err)
	} else {
		fmt.Printf("差戻し後: %s\n", app.Status)
	}

	if err := Resubmit(app, 1); err != nil {
		fmt.Printf("再申請エラー: %v\n", err)
	} else {
		fmt.Printf("再申請後: %s\n", app.Status)
	}

	if err := Approve(app, 2); err != nil {
		fmt.Printf("承認エラー: %v\n", err)
	} else {
		fmt.Printf("承認後: %s\n", app.Status)
	}
}
```
