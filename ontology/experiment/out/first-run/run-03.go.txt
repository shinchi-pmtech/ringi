```go
package main

import (
	"errors"
	"fmt"
)

type Status int

const (
	Draft Status = iota
	Pending
	Approved
	Rejected
)

func (s Status) String() string {
	switch s {
	case Draft:
		return "下書き"
	case Pending:
		return "申請中"
	case Approved:
		return "承認済み"
	case Rejected:
		return "差戻し"
	default:
		return "不明"
	}
}

type User struct {
	ID   string
	Name string
}

type Application struct {
	ID          string
	Applicant   User
	Status      Status
	Title       string
	Description string
}

var (
	ErrInvalidTransition    = errors.New("不正な状態遷移です")
	ErrSameApplicantApprover = errors.New("承認者は申請者と同一人物であってはなりません")
)

func NewApplication(id string, applicant User, title, description string) *Application {
	return &Application{
		ID:          id,
		Applicant:   applicant,
		Status:      Draft,
		Title:       title,
		Description: description,
	}
}

func Submit(app *Application, applicant User) error {
	if app.Applicant.ID != applicant.ID {
		return fmt.Errorf("申請を提出できるのは申請者本人のみです")
	}
	if app.Status != Draft {
		return fmt.Errorf("%w: %s -> %s", ErrInvalidTransition, app.Status, Pending)
	}
	app.Status = Pending
	return nil
}

func Approve(app *Application, approver User) error {
	if app.Status != Pending {
		return fmt.Errorf("%w: %s -> %s", ErrInvalidTransition, app.Status, Approved)
	}
	if app.Applicant.ID == approver.ID {
		return ErrSameApplicantApprover
	}
	app.Status = Approved
	return nil
}

func Reject(app *Application, approver User) error {
	if app.Status != Pending {
		return fmt.Errorf("%w: %s -> %s", ErrInvalidTransition, app.Status, Rejected)
	}
	if app.Applicant.ID == approver.ID {
		return ErrSameApplicantApprover
	}
	app.Status = Rejected
	return nil
}

func main() {
	applicant := User{ID: "user1", Name: "申請者A"}
	approver := User{ID: "user2", Name: "承認者B"}

	app := NewApplication("app1", applicant, "テスト申請", "テスト用の申請です")
	fmt.Printf("初期状態: %s\n", app.Status)

	if err := Submit(app, applicant); err != nil {
		fmt.Printf("提出エラー: %v\n", err)
	} else {
		fmt.Printf("提出後: %s\n", app.Status)
	}

	if err := Approve(app, applicant); err != nil {
		fmt.Printf("承認エラー: %v\n", err)
	}

	if err := Approve(app, approver); err != nil {
		fmt.Printf("承認エラー: %v\n", err)
	} else {
		fmt.Printf("承認後: %s\n", app.Status)
	}

	app2 := NewApplication("app2", applicant, "テスト申請2", "差戻しテスト用")
	_ = Submit(app2, applicant)
	fmt.Printf("\n申請2 提出後: %s\n", app2.Status)

	if err := Reject(app2, approver); err != nil {
		fmt.Printf("差戻しエラー: %v\n", err)
	} else {
		fmt.Printf("差戻し後: %s\n", app2.Status)
	}

	if err := Approve(app2, approver); err != nil {
		fmt.Printf("承認エラー: %v\n", err)
	}
}
```
