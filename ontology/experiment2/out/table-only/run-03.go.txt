// domain/application/application.go
package application

import (
	"fmt"
	"time"
)

// ApplicationID は申請を一意に識別するID
type ApplicationID string

// Application は申請を表すエンティティ
type Application struct {
	id        ApplicationID
	title     string
	content   string
	status    Status
	createdAt time.Time
	updatedAt time.Time
}

// NewApplication は新しい申請を生成する(初期状態は下書き)
func NewApplication(id ApplicationID, title, content string) (*Application, error) {
	if id == "" {
		return nil, fmt.Errorf("申請IDは必須です")
	}
	if title == "" {
		return nil, fmt.Errorf("タイトルは必須です")
	}
	now := time.Now()
	return &Application{
		id:        id,
		title:     title,
		content:   content,
		status:    StatusDraft,
		createdAt: now,
		updatedAt: now,
	}, nil
}

// Reconstruct はDBなどの永続化層から申請を復元する
func Reconstruct(id ApplicationID, title, content, status string, createdAt, updatedAt time.Time) (*Application, error) {
	s, err := NewStatus(status)
	if err != nil {
		return nil, fmt.Errorf("申請の復元に失敗しました: %w", err)
	}
	return &Application{
		id:        id,
		title:     title,
		content:   content,
		status:    s,
		createdAt: createdAt,
		updatedAt: updatedAt,
	}, nil
}

// ID は申請IDを返す
func (a *Application) ID() ApplicationID {
	return a.id
}

// Title はタイトルを返す
func (a *Application) Title() string {
	return a.title
}

// Content は内容を返す
func (a *Application) Content() string {
	return a.content
}

// Status は現在の状態を返す
func (a *Application) Status() Status {
	return a.status
}

// CreatedAt は作成日時を返す
func (a *Application) CreatedAt() time.Time {
	return a.createdAt
}

// UpdatedAt は最終更新日時を返す
func (a *Application) UpdatedAt() time.Time {
	return a.updatedAt
}

// Submit は申請を提出する(下書き → 申請中、または差戻し → 再提出)
func (a *Application) Submit() error {
	if err := a.transition(StatusSubmitted); err != nil {
		return fmt.Errorf("申請の提出に失敗しました: %w", err)
	}
	return nil
}

// Approve は申請を承認する(申請中 → 承認済み)
func (a *Application) Approve() error {
	if err := a.transition(StatusApproved); err != nil {
		return fmt.Errorf("申請の承認に失敗しました: %w", err)
	}
	return nil
}

// Reject は申請を差し戻す(申請中 → 差戻し)
func (a *Application) Reject() error {
	if err := a.transition(StatusRejected); err != nil {
		return fmt.Errorf("申請の差戻しに失敗しました: %w", err)
	}
	return nil
}

// transition は状態遷移の共通処理
func (a *Application) transition(next Status) error {
	if !a.status.CanTransitionTo(next) {
		return fmt.Errorf("状態 %q から %q への遷移は許可されていません", a.status, next)
	}
	a.status = next
	a.updatedAt = time.Now()
	return nil
}
